cmake_minimum_required(VERSION 3.11)

project(CacheSystem LANGUAGES CXX)

set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -std=c++17")

add_executable(cache)

target_sources(cache
	PRIVATE
	src/main.cpp
)

target_include_directories(cache
	PRIVATE
	${CMAKE_SOURCE_DIR}/include
	${CMAKE_SOURCE_DIR}/policies
)

# ---- tests -----------------------------------------------------------------
# Build: cmake --build build        Run: ctest --test-dir build
# Comparison table: ./build/compare
enable_testing()

add_executable(policy_tests tests/PolicyTests.cpp)
target_include_directories(policy_tests PRIVATE
	${CMAKE_SOURCE_DIR}/include
	${CMAKE_SOURCE_DIR}/policies
)
add_test(NAME policy_tests COMMAND policy_tests)

add_executable(compare tests/Compare.cpp)
target_include_directories(compare PRIVATE
	${CMAKE_SOURCE_DIR}/include
	${CMAKE_SOURCE_DIR}/policies
)
